## 一、核心设计原则

**核心理念**：X/Y轴的**表达式**驱动模型的**计算逻辑**，而非固定元素列表

- X轴可以是 `"Ba"`、`"Ba/Eu"`、`"Eu/Eu*"`、`"87Sr/86Sr"`
- 模型必须**解析表达式**，提取所需参数，执行对应计算
- 每个参数组独立计算，独立导出结果

## 二、表达式解析引擎（核心基础设施）

```python
# expression_parser.py
class AxisExpression:
    """解析和计算轴表达式"""
    
    def __init__(self, expression: str):
        """
        表达式类型：
        - "Ba"                    -> 单一元素
        - "Ba/Eu"                 -> 元素比值
        - "Eu/Eu*"                -> Eu异常
        - "87Sr/86Sr"             -> 同位素比值
        - "[Ba]+[Eu]"             -> 元素和
        - "[Ba]*[Eu]"             -> 元素积
        - "自定义元素"            -> 用户自定义元素
        """
        self.raw_expr = expression
        self.parsed = self._parse_expression()
    
    def _parse_expression(self) -> Dict[str, Any]:
        """解析表达式结构"""
        # 类型1：单一元素
        if self.raw_expr.isalpha() and self.raw_expr[0].isupper():
            return {
                "type": "element_concentration",
                "element": self.raw_expr,
                "required_params": [f"D_{self.raw_expr}", f"C0_{self.raw_expr}"]
            }
        
        # 类型2：同位素比值（如87Sr/86Sr）
        if "/" in self.raw_expr and all(c.isdigit() or c.isalpha() for c in self.raw_expr.replace("/", "")):
            numerator, denominator = self.raw_expr.split("/")
            return {
                "type": "isotope_ratio",
                "numerator": numerator,
                "denominator": denominator,
                "required_params": [f"IC0_{numerator}", f"IC0_{denominator}"]
            }
        
        # 类型3：元素比值（如Ba/Eu）
        if "/" in self.raw_expr and self.raw_expr.replace("/", "").isalpha():
            numerator, denominator = self.raw_expr.split("/")
            return {
                "type": "element_ratio",
                "numerator": numerator,
                "denominator": denominator,
                "required_params": [
                    f"D_{numerator}", f"C0_{numerator}",
                    f"D_{denominator}", f"C0_{denominator}"
                ]
            }
        
        # 类型4：Eu异常 (Eu/Eu*)
        if self.raw_expr == "Eu/Eu*":
            return {
                "type": "eu_anomaly",
                "required_params": [
                    "D_Eu", "C0_Eu",
                    "D_Sm", "C0_Sm",
                    "D_Gd", "C0_Gd"
                ],
                "elements": ["Eu", "Sm", "Gd"]
            }
        
        # 类型5：Ce异常 (Ce/Ce*)
        if self.raw_expr == "Ce/Ce*":
            return {
                "type": "ce_anomaly",
                "required_params": [
                    "D_Ce", "C0_Ce",
                    "D_La", "C0_La",
                    "D_Pr", "C0_Pr"
                ],
                "elements": ["Ce", "La", "Pr"]
            }
        
        # 类型6：元素运算（如[Ba]+[Eu]）
        if "[" in self.raw_expr:
            # 使用正则提取所有元素
            import re
            elements = re.findall(r'\[([A-Za-z]+)\]', self.raw_expr)
            return {
                "type": "element_operation",
                "expression": self.raw_expr,
                "elements": elements,
                "required_params": [f"D_{e}" for e in elements] + [f"C0_{e}" for e in elements]
            }
        
        # 类型7：自定义元素
        return {
            "type": "custom_element",
            "element": self.raw_expr,
            "is_user_defined": True,
            "required_params": [f"D_{self.raw_expr}", f"C0_{self.raw_expr}"]
        }
    
    def get_required_parameters(self) -> List[str]:
        """返回计算此表达式所需的所有参数名称"""
        return self.parsed["required_params"]
    
    def evaluate(self, computed_values: Dict[str, float]) -> float:
        """
        根据已计算的值求表达式结果
        computed_values: 模型已计算出的各元素值
        """
        expr_type = self.parsed["type"]
        
        if expr_type == "element_concentration":
            element = self.parsed["element"]
            return computed_values.get(element, 0.0)
        
        elif expr_type == "isotope_ratio":
            num = self.parsed["numerator"]
            den = self.parsed["denominator"]
            return computed_values.get(num, 0.0) / computed_values.get(den, 1.0)
        
        elif expr_type == "element_ratio":
            num = self.parsed["numerator"]
            den = self.parsed["denominator"]
            return computed_values.get(num, 0.0) / computed_values.get(den, 1.0)
        
        elif expr_type == "eu_anomaly":
            Eu = computed_values.get("Eu", 0.0)
            Sm = computed_values.get("Sm", 0.0)
            Gd = computed_values.get("Gd", 0.0)
            # Eu/Eu* = Eu_N / sqrt(Sm_N * Gd_N)
            # 这里简化处理，实际需归一化
            return Eu / ((Sm * Gd) ** 0.5)
        
        elif expr_type == "ce_anomaly":
            Ce = computed_values.get("Ce", 0.0)
            La = computed_values.get("La", 0.0)
            Pr = computed_values.get("Pr", 0.0)
            return Ce / ((La * Pr) ** 0.5)
        
        elif expr_type == "element_operation":
            # 使用eval计算表达式，已计算的值替换元素名
            expr = self.parsed["expression"]
            for elem in self.parsed["elements"]:
                value = computed_values.get(elem, 0.0)
                expr = expr.replace(f"[{elem}]", str(value))
            return eval(expr)
        
        return 0.0
```

## 三、重构的模型基类

```python
# model_base.py
from abc import ABC, abstractmethod
from typing import Dict, List, Tuple, Any
import pandas as pd
import numpy as np

class ModelBase(ABC):
    """所有模型的抽象基类"""
    
    @abstractmethod
    def get_metadata(self) -> Dict[str, Any]:
        """返回模型元数据"""
        pass
    
    def get_required_parameters(self, x_axis: 'AxisExpression', y_axis: 'AxisExpression') -> List[Dict[str, Any]]:
        """
        根据X/Y轴表达式动态生成参数定义
        这是动态UI生成的核心
        """
        # 提取X和Y轴所需的所有参数
        x_params = x_axis.get_required_parameters() if x_axis else []
        y_params = y_axis.get_required_parameters() if y_axis else []
        
        all_params = list(set(x_params + y_params))
        
        # 转换为详细定义
        param_definitions = []
        for param_name in all_params:
            param_definitions.append(self._create_param_definition(param_name))
        
        # 添加模型特定参数（如F, r等）
        param_definitions.extend(self._get_model_specific_params())
        
        return param_definitions
    
    def _create_param_definition(self, param_name: str) -> Dict[str, Any]:
        """根据参数名称猜参数类型并创建定义"""
        if param_name.startswith("D_"):
            element = param_name[2:]
            return {
                "name": param_name,
                "display_name": f"分配系数 D({element})",
                "type": "float",
                "default": 1.0,
                "step": 0.1,
                "min": 0.01,
                "max": 50.0,
                "unit": "无量纲",
                "category": "partition_coefficient"
            }
        elif param_name.startswith("C0_"):
            element = param_name[3:]
            return {
                "name": param_name,
                "display_name": f"初始浓度 C₀({element})",
                "type": "float",
                "default": 100.0,
                "step": 1.0,
                "min": 0,
                "max": 10000,
                "unit": "ppm",
                "category": "initial_composition"
            }
        elif param_name.startswith("IC0_"):
            isotope = param_name[4:]
            return {
                "name": param_name,
                "display_name": f"初始同位素 {isotope}",
                "type": "float",
                "default": 0.704 if "Sr" in isotope else 0.512,
                "step": 0.001,
                "min": 0.5,
                "max": 1.0,
                "unit": "‰",
                "category": "isotope"
            }
        # ... 其他参数类型
        
        return {"name": param_name, "type": "float", "default": 1.0}
    
    @abstractmethod
    def _get_model_specific_params(self) -> List[Dict[str, Any]]:
        """返回模型特有参数定义"""
        pass
    
    def calculate(self, 
                  parameters: Dict[str, float],
                  x_axis: 'AxisExpression',
                  y_axis: 'AxisExpression',
                  variable_param: str = 'F',
                  variable_range: np.ndarray = None) -> pd.DataFrame:
        """
        核心计算方法：根据轴表达式计算曲线
        
        Parameters
        ----------
        parameters : 参数字典
        x_axis : X轴表达式
        y_axis : Y轴表达式
        variable_param : 变化参数（通常是F或N）
        variable_range : 参数变化范围数组
        
        Returns
        -------
        DataFrame: {'x': [...], 'y': [...], 'variable': [...], 'details': {...}}
        """
        if variable_range is None:
            variable_range = np.linspace(0.1, 1.0, 30)
        
        results = []
        for var_val in variable_range:
            # 创建临时参数集
            temp_params = parameters.copy()
            if variable_param in temp_params:
                temp_params[variable_param] = var_val
            
            # 计算所有需要的元素和同位素值
            computed_values = self._compute_all_needed_values(temp_params, x_axis, y_axis)
            
            # 使用AxisExpression计算最终的X和Y值
            x_val = x_axis.evaluate(computed_values) if x_axis else 0.0
            y_val = y_axis.evaluate(computed_values) if y_axis else 0.0
            
            results.append({
                'x': x_val,
                'y': y_val,
                'variable': var_val,
                'details': computed_values  # 存储所有中间计算结果
            })
        
        return pd.DataFrame(results)
    
    @abstractmethod
    def _compute_all_needed_values(self, parameters: Dict[str, float], 
                                   x_axis: 'AxisExpression',
                                   y_axis: 'AxisExpression') -> Dict[str, float]:
        """
        计算X和Y表达式所需的所有中间值
        这是模型具体实现的核心
        """
        pass
    
    def export_group_results(self, group_name: str, parameters: Dict[str, float],
                           x_axis: 'AxisExpression', y_axis: 'AxisExpression',
                           df_results: pd.DataFrame) -> pd.DataFrame:
        """
        导出参数组完整结果
        包含所有中间计算值、参数、元数据
        """
        export_data = {
            '参数组名称': group_name,
            '模型类型': self.get_metadata()['name'],
            **{k: v for k, v in parameters.items()},
            'X轴表达式': x_axis.raw_expr if x_axis else 'None',
            'Y轴表达式': y_axis.raw_expr if y_axis else 'None',
        }
        
        # 添加计算结果
        result_table = df_results.copy()
        result_table['参数组'] = group_name
        
        # 展开details列
        details_df = pd.DataFrame(result_table['details'].tolist())
        result_table = pd.concat([result_table.drop('details', axis=1), details_df], axis=1)
        
        return result_table
```

```python
# models/fc_model.py
class FCModel(ModelBase):
    def get_metadata(self):
        return {
            "name": "FC模型",
            "description": "分离结晶（Fractional Crystallization）",
            "category": "magmatic",
            "formula": "C^l = C0 * F^(D-1), 同位素不变"
        }
    
    def _get_model_specific_params(self):
        return [
            {
                "name": "F",
                "display_name": "残余熔体分数 F",
                "type": "float",
                "default": 0.8,
                "step": 0.01,
                "min": 0.01,
                "max": 1.0,
                "unit": "分数",
                "category": "process_control"
            }
        ]
    
    def _compute_all_needed_values(self, parameters, x_axis, y_axis):
        """
        FC模型计算逻辑
        1. 提取表达式涉及的所有元素
        2. 对每个元素计算C = C0 * F^(D-1)
        3. 同位素值保持不变
        """
        computed = {}
        F = parameters['F']
        
        # 获取X和Y轴涉及的所有元素和同位素
        all_elements = set()
        all_isotopes = set()
        
        if x_axis:
            if x_axis.parsed['type'] == 'element_concentration':
                all_elements.add(x_axis.parsed['element'])
            elif x_axis.parsed['type'] == 'element_ratio':
                all_elements.add(x_axis.parsed['numerator'])
                all_elements.add(x_axis.parsed['denominator'])
            elif x_axis.parsed['type'] in ['eu_anomaly', 'ce_anomaly']:
                all_elements.update(x_axis.parsed['elements'])
            elif x_axis.parsed['type'] == 'isotope_ratio':
                all_isotopes.add(x_axis.parsed['numerator'])
                all_isotopes.add(x_axis.parsed['denominator'])
        
        if y_axis:
            # 同样处理Y轴
            if y_axis.parsed['type'] == 'element_concentration':
                all_elements.add(y_axis.parsed['element'])
            elif y_axis.parsed['type'] == 'element_ratio':
                all_elements.add(y_axis.parsed['numerator'])
                all_elements.add(y_axis.parsed['denominator'])
            elif y_axis.parsed['type'] in ['eu_anomaly', 'ce_anomaly']:
                all_elements.update(y_axis.parsed['elements'])
            elif y_axis.parsed['type'] == 'isotope_ratio':
                all_isotopes.add(y_axis.parsed['numerator'])
                all_isotopes.add(y_axis.parsed['denominator'])
            elif y_axis.parsed['type'] == 'element_operation':
                all_elements.update(y_axis.parsed['elements'])
        
        # 计算元素浓度
        for elem in all_elements:
            D_key = f"D_{elem}"
            C0_key = f"C0_{elem}"
            
            D = parameters.get(D_key, 1.0)
            C0 = parameters.get(C0_key, 100.0)
            
            # FC核心公式
            C = C0 * (F ** (D - 1))
            computed[elem] = C
        
        # 同位素值（FC不改变同位素）
        for iso in all_isotopes:
            IC0_key = f"IC0_{iso}"
            IC0 = parameters.get(IC0_key, 0.704)
            computed[iso] = IC0  # 保持不变
        
        return computed
    
    def validate_parameters(self, parameters):
        F = parameters.get('F', 0)
        if not (0.01 <= F <= 1.0):
            return False, "F值必须在0.01到1.0之间"
        return True, ""
```

```python
# models/afc_model.py
class AFCModel(ModelBase):
    def get_metadata(self):
        return {
            "name": "AFC模型",
            "description": "同化-分离结晶（Assimilation Fractional Crystallization）",
            "category": "magmatic",
            "formula": "C^l = C0·(1-F-Z)^(-r/(r+D-1)) + ...",
            "constraints": "开放系统，同位素改变"
        }
    
    def _get_model_specific_params(self):
        return [
            {
                "name": "F",
                "display_name": "残余熔体分数 F",
                "type": "float",
                "default": 0.7,
                "step": 0.01,
                "min": 0.01,
                "max": 1.0,
                "unit": "分数",
                "category": "process_control"
            },
            {
                "name": "r",
                "display_name": "同化/结晶速率比 r",
                "type": "float",
                "default": 0.2,
                "step": 0.05,
                "min": 0.01,
                "max": 10.0,
                "unit": "无量纲",
                "category": "process_control"
            }
        ]
    
    def _compute_all_needed_values(self, parameters, x_axis, y_axis):
        """
        AFC模型计算逻辑
        复杂公式，需要计算每个元素的浓度和同位素
        """
        computed = {}
        F = parameters['F']
        r = parameters['r']
        
        # 收集所有需要的元素和同位素
        all_elements = set()
        all_isotopes = set()
        
        for axis in [x_axis, y_axis]:
            if not axis:
                continue
            if axis.parsed['type'] == 'element_concentration':
                all_elements.add(axis.parsed['element'])
            elif axis.parsed['type'] == 'element_ratio':
                all_elements.add(axis.parsed['numerator'])
                all_elements.add(axis.parsed['denominator'])
            elif axis.parsed['type'] in ['eu_anomaly', 'ce_anomaly']:
                all_elements.update(axis.parsed['elements'])
            elif axis.parsed['type'] == 'element_operation':
                all_elements.update(axis.parsed['elements'])
            elif axis.parsed['type'] == 'isotope_ratio':
                all_isotopes.add(axis.parsed['numerator'])
                all_isotopes.add(axis.parsed['denominator'])
        
        # 计算每个元素的浓度和同位素
        for elem in all_elements:
            D = parameters.get(f"D_{elem}", 1.0)
            C0 = parameters.get(f"C0_{elem}", 100.0)
            Ca = parameters.get(f"Ca_{elem}", 50.0)
            
            # 计算Z值
            if abs(r - 1.0) < 1e-6:  # r≈1的特殊情况
                # 使用极限形式
                Z = 1.0  # 简化处理
                term1 = C0 * np.exp(-r * (1 - F) / (1 - F))  # 需要更精确的极限公式
            else:
                Z = (r + D - 1) / (r - 1)
                term1 = C0 * ((1 - F - Z) ** (-r / (r + D - 1)))
                term2 = (r / (r + D - 1)) * Ca * (1 - ((1 - F - Z) ** ((r + D - 1) / (r - 1))))
                C = term1 + term2
                computed[elem] = C
        
        # 计算同位素
        for iso in all_isotopes:
            D_iso = parameters.get(f"D_{iso}", 1.0)
            if abs(r - 1.0) < 1e-6:
                # r≈1的极限情况
                Z_iso = 1.0
                IC = parameters.get(f"IC0_{iso}", 0.704)  # 简化
            else:
                Z_iso = (r + D_iso - 1) / (r - 1)
                IC0 = parameters.get(f"IC0_{iso}", 0.704)
                ICa = parameters.get(f"ICa_{iso}", 0.720)
                C0 = parameters.get(f"C0_{iso}", 100.0)
                Ca = parameters.get(f"Ca_{iso}", 50.0)
                
                numerator = (r / (r - 1)) * Ca * (1 - F - Z_iso) * ICa + C0 * F * Z_iso * IC0
                denominator = (r / (r - 1)) * Ca * (1 - F - Z_iso) + C0 * F * Z_iso
                IC = numerator / denominator
            
            computed[iso] = IC
        
        return computed
    
    def validate_parameters(self, parameters):
        F = parameters.get('F', 0)
        r = parameters.get('r', 0)
        if F <= 0 or F > 1:
            return False, "F值必须在0到1之间"
        if r <= 0:
            return False, "r值必须大于0"
        return True, ""
```

```python
# models/fca_model.py
class FCAModel(ModelBase):
    def get_metadata(self):
        return {
            "name": "FCA模型",
            "description": "解耦同化-分离结晶",
            "category": "magmatic",
            "formula": "C^l = (Ca·r·Mc + Cf·(1-Mc))/F",
            "constraints": "F > r"
        }
    
    def _get_model_specific_params(self):
        return [
            {
                "name": "F",
                "display_name": "残余熔体分数 F",
                "default": 0.7,
                "step": 0.01,
                "min": 0.01,
                "max": 1.0,
                "constraints": "> r"
            },
            {
                "name": "r",
                "display_name": "同化/结晶速率比 r",
                "default": 0.3,
                "step": 0.05,
                "min": 0.01,
                "max": 0.9  # 必须小于F的最大值
            }
        ]
    
    def _compute_all_needed_values(self, parameters, x_axis, y_axis):
        computed = {}
        F = parameters['F']
        r = parameters['r']
        
        # 验证约束
        if F <= r:
            raise ValueError("FCA模型要求 F > r")
        
        # 计算Mc
        Mc = 1 - (F - r) / (1 - r)
        
        # 收集元素
        elements = self._extract_elements_from_axes([x_axis, y_axis])
        
        for elem in elements:
            D = parameters.get(f"D_{elem}", 1.0)
            C0 = parameters.get(f"C0_{elem}", 100.0)
            Ca = parameters.get(f"Ca_{elem}", 50.0)
            
            # 计算Cf (纯FC结果)
            Cf = C0 * (F ** (D - 1))
            
            # FCA浓度
            C = (Ca * r * Mc + Cf * (1 - Mc)) / F
            computed[elem] = C
        
        return computed
```

```python
                "min": 1.0,
                "max": 1.02
            }
        ]
    
    def _compute_all_needed_values(self, parameters, x_axis, y_axis):
        computed = {}
        F = parameters['F']
        alpha = parameters.get('alpha', 1.008)
        
        # 处理元素浓度
        elements = set()
        isotopes = set()
        
        for axis in [x_axis, y_axis]:
            if not axis:
                continue
            if axis.parsed['type'] == 'element_concentration':
                elements.add(axis.parsed['element'])
            elif axis.parsed['type'] == 'isotope_ratio':
                isotopes.add(axis.parsed['numerator'])
        
        # 元素浓度（Rayleigh分馏）
        for elem in elements:
            D = parameters.get(f"D_{elem}", 1.4)
            C0 = parameters.get(f"C0_{elem}", 100.0)
            C = C0 * (F ** (D - 1))
            computed[elem] = C
        
        # 同位素（Rayleigh公式）
        for iso in isotopes:
            delta_i = parameters.get(f"delta_i_{iso}", 0.0)
            delta_f = delta_i + 1000 * (F  ** (alpha - 1) - 1)
            computed[iso] = delta_f
        
        return computed
```

```python
class ModelFactory:
    """模型工厂：统一管理所有模型类"""
    
    # 模型注册表
    _model_registry: Dict[str, Type[ModelBase]] = {}
    
    @classmethod
    def register_model(cls, model_name: str, model_class: Type[ModelBase]):
        """注册新模型"""
        cls._model_registry[model_name] = model_class
    
    @classmethod
    def get_model_names(cls) -> list:
        """返回所有可用模型名称"""
        return list(cls._model_registry.keys())
    
    @classmethod
    def create_model(cls, model_name: str) -> ModelBase:
        """创建模型实例"""
        if model_name not in cls._model_registry:
            raise ValueError(f"未知模型: {model_name}")
        return cls._model_registry[model_name]()
    
    @classmethod
    def get_all_metadata(cls) -> Dict[str, Dict]:
        """获取所有模型元数据"""
        metadata = {}
        for name, model_class in cls._model_registry.items():
            instance = model_class()
            metadata[name] = instance.get_metadata()
        return metadata
```

```python
from PyQt6.QtCore import QObject, pyqtSignal
import pandas as pd
import numpy as np

class ParameterGroup(QObject):
    """参数组：代表一个模型图层"""
    
    dataChanged = pyqtSignal()
    curveCalculated = pyqtSignal(pd.DataFrame)
    
    def __init__(self, group_id: str, model_name: str, model: ModelBase):
        super().__init__()
        self.group_id = group_id
        self.model_name = model_name
        self.model = model
        self.name = f"组_{group_id}_{model_name}"
        self.visible = True
        self.color = "#FF0000"
        self.linestyle = "-"
        self.parameters = {}
        self.x_axis_expr = None
        self.y_axis_expr = None
        
        # 初始化默认参数
        self.reset_parameters()
    
    def reset_parameters(self):
        """重置为模型默认参数"""
        # 根据当前轴表达式获取参数定义
        param_defs = self.model.get_required_parameters(self.x_axis_expr, self.y_axis_expr)
        self.parameters = {p['name']: p['default'] for p in param_defs}
        self.dataChanged.emit()
    
    def set_axis_expressions(self, x_expr: 'AxisExpression', y_expr: 'AxisExpression'):
        """设置轴表达式并更新参数"""
        self.x_axis_expr = x_expr
        self.y_axis_expr = y_expr
        
        # 重新计算所需参数
        param_defs = self.model.get_required_parameters(x_expr, y_expr)
        
        # 保留已有参数值，新增参数用默认值
        new_params = {}
        for p_def in param_defs:
            name = p_def['name']
            if name in self.parameters:
                new_params[name] = self.parameters[name]
            else:
                new_params[name] = p_def['default']
        
        self.parameters = new_params
        self.dataChanged.emit()
    
    def set_parameter(self, param_name: str, value: float):
        """设置单个参数"""
        self.parameters[param_name] = value
        self.dataChanged.emit()
    
    def calculate_curve(self, variable_range: np.ndarray = None) -> pd.DataFrame:
        """计算曲线数据"""
        if variable_range is None:
            variable_range = np.linspace(0.1, 1.0, 50)
        
        # 判断变化参数（通常是F或N）
        variable_param = 'F' if 'F' in self.parameters else 'N'
        
        # 调用模型计算
        df_results = self.model.calculate(
            self.parameters,
            self.x_axis_expr,
            self.y_axis_expr,
            variable_param,
            variable_range
        )
        
        # 添加元数据
        df_results['group_id'] = self.group_id
        df_results['group_name'] = self.name
        df_results['model_name'] = self.model_name
        
        return df_results
    
    def export_results(self) -> pd.DataFrame:
        """导出该参数组的完整结果"""
        if not hasattr(self, '_last_results'):
            return pd.DataFrame()
        
        return self.model.export_group_results(
            self.name,
            self.parameters,
            self.x_axis_expr,
            self.y_axis_expr,
            self._last_results
        )

class ParameterGroupManager(QObject):
    """管理所有参数组（图层）"""
    
    groupsChanged = pyqtSignal()
    curveDataChanged = pyqtSignal(pd.DataFrame)
    
    def __init__(self):
        super().__init__()
        self.groups: List[ParameterGroup] = []
        self._next_id = 1
    
    def add_group(self, model_name: str) -> ParameterGroup:
        """添加新参数组"""
        from .model_factory import ModelFactory
        
        model = ModelFactory.create_model(model_name)
        group = ParameterGroup(f"g{self._next_id}", model_name, model)
        self.groups.append(group)
        self._next_id += 1
        
        group.dataChanged.connect(self.on_group_data_changed)
        self.groupsChanged.emit()
        return group
    
    def remove_group(self, group_id: str):
        """删除参数组"""
        self.groups = [g for g in self.groups if g.group_id != group_id]
        self.groupsChanged.emit()
        self.curveDataChanged.emit(pd.DataFrame())
    
    def get_visible_groups(self) -> List[ParameterGroup]:
        """获取可见参数组"""
        return [g for g in self.groups if g.visible]
    
    def update_all_group_axes(self, x_expr: 'AxisExpression', y_expr: 'AxisExpression'):
        """轴改变时更新所有参数组"""
        for group in self.groups:
            group.set_axis_expressions(x_expr, y_expr)
    
    def export_all_groups(self, filepath: str):
        """导出所有参数组结果"""
```

```python
                self.parameterChanged.emit(p_def['name'], value)
                self.sender().setStyleSheet("")
            else:
```

```python
        self.param_group_mgr.curveDataChanged.connect(self.on_plot_update)
        
        # 图解交互
        self.plot_widget.pointClicked.connect(self.on_plot_point_clicked)
    
    def on_axes_changed(self, x_expr: AxisExpression, y_expr: AxisExpression):
        """轴配置改变"""
        self.x_axis_expr = x_expr
        self.y_axis_expr = y_expr
        
        # 更新所有参数组的轴配置
        self.param_group_mgr.update_all_group_axes(x_expr, y_expr)
        
        # 重新加载当前选中的参数组面板
        current_item = self.group_list_widget.currentItem()
        if current_item:
            group_id = current_item.data(Qt.UserRole)
            group = next((g for g in self.param_group_mgr.groups if g.group_id == group_id), None)
            if group:
                self.param_panel.load_parameter_group(group)
        
        # 更新图解
        self.on_plot_update()
    
    def on_add_group(self):
        """添加参数组"""
        model_names = self.model_factory.get_model_names()
        
        # 弹出模型选择对话框
        dialog = QDialog(self)
        layout = QVBoxLayout()
        
        layout.addWidget(QLabel("选择模型类型:"))
        model_combo = QComboBox()
        model_combo.addItems(model_names)
        layout.addWidget(model_combo)
        
        buttons = QHBoxLayout()
        ok_btn = QPushButton("确定")
        cancel_btn = QPushButton("取消")
        buttons.addWidget(ok_btn)
        buttons.addWidget(cancel_btn)
        layout.addLayout(buttons)
        
        dialog.setLayout(layout)
        
        ok_btn.clicked.connect(dialog.accept)
        cancel_btn.clicked.connect(dialog.reject)
        
        if dialog.exec() == QDialog.Accepted:
            model_name = model_combo.currentText()
            
            # 创建参数组
            group = self.param_group_mgr.add_group(model_name)
            
            # 添加到列表
            item = QListWidgetItem(group.name)
            item.setData(Qt.UserRole, group.group_id)
            item.setFlags(item.flags() | Qt.ItemIsEditable)
            self.group_list_widget.addItem(item)
            
            # 自动选中新参数组
            self.group_list_widget.setCurrentItem(item)
    
    def on_remove_group(self):
        """删除当前参数组"""
        current_item = self.group_list_widget.currentItem()
        if current_item:
            group_id = current_item.data(Qt.UserRole)
            self.param_group_mgr.remove_group(group_id)
            self.group_list_widget.takeItem(self.group_list_widget.row(current_item))
    
    def on_duplicate_group(self):
        """复制当前参数组"""
        current_item = self.group_list_widget.currentItem()
        if not current_item:
            return
        
        # 获取当前组
        group_id = current_item.data(Qt.UserRole)
        group = next((g for g in self.param_group_mgr.groups if g.group_id == group_id), None)
        
        if group:
            # 创建新组
            new_group = self.param_group_mgr.add_group(group.model_name)
            new_group.name = f"{ group.name}_copy"
            new_group.parameters = group.parameters.copy()
            new_group.color = group.color
            new_group.linestyle = group.linestyle
            
            # 添加到列表
            item = QListWidgetItem(new_group.name)
            item.setData(Qt.UserRole, new_group.group_id)
            self.group_list_widget.addItem(item)
            
            self.group_list_widget.setCurrentItem(item)
    
    def on_group_selected(self, current, previous):
        """选中参数组时加载其面板"""
        if not current:
            return
        
        group_id = current.data(Qt.UserRole)
        group = next((g for g in self.param_group_mgr.groups if g.group_id == group_id), None)
        
        if group:
            self.param_panel.load_parameter_group(group)
            # 显示该组的计算结果
            self.result_widget.display_group_results(group)
    
    def on_parameter_changed(self, param_name: str, value: float):
        """参数值改变"""
        current_item = self.group_list_widget.currentItem()
        if not current_item:
            return
        
        group_id = current_item.data(Qt.UserRole)
        group = next((g for g in self.param_group_mgr.groups if g.group_id == group_id), None)
        
        if group:
            group.set_parameter(param_name, value)
    
    def on_plot_update(self):
        """更新图解"""
        # 获取可见参数组
        visible_groups = self.param_group_mgr.get_visible_groups()
        
        if not visible_groups:
            self.plot_widget.clear()
            return
        
        # 并行计算所有曲线
        curves = []
        for group in visible_groups:
            df = group.calculate_curve()
            
            curves.append({
                'x': df['x'].values,
                'y': df['y'].values,
                'variable': df['variable'].values,
                'name': group.name,
                'color': group.color,
                'linestyle': group.linestyle
            })
            
            # 保存结果供导出
            group._last_results = df
        
        # 绘制
        self.plot_widget.plot_multiple_curves(curves)
        
        # 更新结果面板
        if visible_groups:
            self.result_widget.display_group_results(visible_groups[0])
    
    def on_plot_point_clicked(self, x, y):
        """图解点点击事件"""
        # 显示该点的详细信息
        self.result_widget.show_point_details(x, y)

class ResultPanel(QWidget):
    """计算结果面板"""
    
    def __init(self, parent=None):
        super().__init__(parent)
        self.current_group = None
        
        layout = QVBoxLayout()
        
        # 标题
        self.title_label = QLabel("请选择一个参数组")
        layout.addWidget(self.title_label)
        
        # 结果表格
        self.result_table = QTableWidget()
        self.result_table.setAlternatingRowColors(True)
        layout.addWidget(self.result_table)
        
        # 导出按钮
        self.export_btn = QPushButton("导出当前组")
        self.export_btn.clicked.connect(self.export_current_group)
        layout.addWidget(self.export_btn)
        
        self.setLayout(layout)
    
    def display_group_results(self, group: ParameterGroup):
        """显示参数组计算结果"""
        if not group or not hasattr(group, '_last_results'):
            return
        
        self.current_group = group
        self.title_label.setText(f"<b>{group.name}</b> - 计算结果")
        
        df = group._last_results
        
        # 设置表格
        self.result_table.setRowCount(len(df))
        self.result_table.setColumnCount(len(df.columns))
        self.result_table.setHorizontalHeaderLabels(df.columns)
        
        for i, row in df.iterrows():
            for j, col in enumerate(df.columns):
                item = QTableWidgetItem(str(row[col]))
                self.result_table.setItem(i, j, item)
        
        self.result_table.resizeColumnsToContents()
    
    def export_current_group(self):
        """导出当前参数组结果"""
        if not self.current_group:
            return
        
        from PyQt6.QtWidgets import QFileDialog
        filepath, _ = QFileDialog.getSaveFileName(
            self,
            "导出结果",
            f"{self.current_group.name}_results.csv",
            "CSV Files (*.csv)"
        )
        
        if filepath:
            df = self.current_group.export_results()
            df.to_csv(filepath, index=False)
```

## 十、完整数据流总结

```
1. 用户加载Excel文件
   ↓
2. 用户配置X/Y轴表达式 (AxisExpression解析)
   ↓
3. 用户添加参数组 → ModelFactory创建模型实例
   ↓
4. 参数组根据轴表达式生成参数定义
   ↓
5. UI动态生成参数控件 (DynamicParamPanel)
   ↓
6. 用户调整参数 → ParameterGroup.set_parameter()
   ↓
7. 触发calculate_curve() → 模型._compute_all_needed_values()
   ↓
8. AxisExpression.evaluate() 计算最终X/Y值
   ↓
9. PlotWidget绘制多条曲线
   ↓
10. ResultPanel显示结果表格
    ↓
11. 用户点击"导出" → export_group_results() 生成CSV
```

**核心优势**：
- **表达式驱动**：X/Y轴任意组合，模型自动适配
- **模块完全独立**：新增模型只需继承ModelBase，注册到工厂
- **参数动态生成**：根据轴表达式自动提取所需参数
- **精准计算**：每个模型都实现了完整的计算逻辑，支持所有轴类型
- **灵活导出**：每个参数组可独立导出完整计算过程和结果
